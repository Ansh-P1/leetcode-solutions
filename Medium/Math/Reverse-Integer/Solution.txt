class Solution {
public:
    int reverse(int x) {
        int a = 1;
        if (x<0){
            a = -1;
        }
        long long ab = llabs((long long) x);
        long long ans =0;
        while(ab>0){
            int y = ab%10;
            ans = ans*10 + y;
            ab = ab/10;
        }

        ans = ans *a;
        if (ans>INT_MAX || ans<INT_MIN) return 0;
        return ans;
        
    }
};
