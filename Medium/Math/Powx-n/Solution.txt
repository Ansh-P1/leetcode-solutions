class Solution {
public:
    double myPow(double x, int n) {
        long long a = abs((long long)n);
        double ans = 1;

        while (a > 0){
            if (a % 2 == 1){
                ans = ans * x;
            }
            x = x * x;
            a = a / 2;
        }

        if (n < 0){
            return 1.0 / ans;
        }
        return ans;
    }
};
