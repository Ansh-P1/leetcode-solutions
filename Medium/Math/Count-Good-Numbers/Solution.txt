
class Solution {
public:
    long long power(long long base, long long exp, long long mod){
        long long result = 1;
        base = base % mod;
        while (exp > 0){
            if (exp % 2 == 1){
                result = (result * base) % mod;
            }
            base = (base * base) % mod;
            exp = exp / 2;
        }
        return result;
    }
    int countGoodNumbers(long long n) {
        long long odd = n/2;
        long long even = n/2;
        if(n%2 ==1){
            even = even +1;
        }
        long long MOD = 1e9+7;
        long long ans = (power(5, even, MOD) * power(4, odd, MOD)) % MOD;
        return (int) ans;
    }
};
