/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        if (l1 == NULL && l2 == NULL){
            return NULL;
        }
        if(l1 == NULL && l2!=NULL){
            return l2;
        }
        if(l2 == NULL && l1!=NULL){
            return l1;
        }

        ListNode* dum = new ListNode(0);
        ListNode* curr = dum;

        int rem =0;
        ListNode* a = l1;
        ListNode* b = l2;

        while(a != NULL || b != NULL || rem!=0){
            int x =0;
            int y =0;
            if (a != NULL){
                x = a->val;
            }
            if (b != NULL){
                y = b->val;
            }
            int plus = x+y+ rem;
            curr->next = new ListNode ( plus%10);
            curr = curr->next;
            rem= plus/10;
            if(a!=NULL){
                a = a->next;
            }
            if (b!=NULL){
                b = b->next;
            }
        }
        return dum->next;
        
    }
};
