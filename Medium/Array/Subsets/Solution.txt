class Solution {
public:
    void solve(int ind, vector<int>& arr, vector<vector<int>> &ans, vector<int>& ds ){
        if (ind == arr.size()){
            ans.push_back(ds);
            return;
        }

        solve(ind+1,arr,ans,ds);

        ds.push_back(arr[ind]);
        solve(ind+1, arr, ans,ds);
        ds.pop_back();
    }
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> ans;
        vector<int> ds;
        solve(0,nums,ans,ds);
        return ans;
        
    }
};
