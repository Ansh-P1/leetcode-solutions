class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int n = matrix.size();
        int m = matrix[0].size();
        int ans =-1;
        for (int i=0; i<n; i++){
            if (matrix[i][m-1] < target || matrix[i][0]>target){
                continue;
            }
            int low = 0;
            int high = m-1;
            while (low <= high){
            

                int mid = (low + high)/2;
                if (matrix[i][mid] > target){
                        high = mid-1;
                }
                else if (matrix[i][mid]== target) {
                    ans = mid;
                    break;
                }
                else {
                    low = mid +1;
                }
            }

        }
        if (ans != -1){
            return true;
        }
        else return false;
    }
};
