class Solution {
public:
    void solve(int start, int k , int n, int sum, vector<vector<int>>& ans, vector<int>& ds){
        if (ds.size() == k){
            if(sum == n){
                ans.push_back(ds);
                
            }
            return;
        }
        if (sum >= n){
            return;
        }
        for(int i=start; i<=9; i++){
            ds.push_back(i);
            solve(i+1, k, n,sum+i, ans, ds);
            ds.pop_back();
        }

    }
    vector<vector<int>> combinationSum3(int k, int n) {
        vector<vector<int>> ans;
        vector<int> ds;
        solve(1,k,n,0,ans,ds);
        return ans;
        
    }
};
